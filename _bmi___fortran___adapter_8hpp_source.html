330 get_analogous_cxx_type(const std::string &external_type_name, const size_t item_size) override {
331 const bool is_integer = external_type_name == "integer" || external_type_name == "int";
332 const bool is_single = external_type_name == "real" || external_type_name == "float";
333 const bool is_double = external_type_name == "double precision" || external_type_name == "double";
334
335 if (is_integer) {
336 switch (item_size) {
337 case 1: return "int8_t";
338 case 2: return "int16_t";
339 case 4: return "int32_t";
340 case 8: return "int64_t";
341 }
342 }
343 if (is_single || is_double) {
344 if (is_single && item_size == sizeof(float)) {return "float";}
345 if (item_size == sizeof(double)) {return "double";}
346 // IEEE binary128 is the one format with a 113-bit significand; x87's 80-bit type has 64, and
347 // PowerPC's double-double 106.
348 constexpr bool long_double_is_float128 = std::numeric_limits<long double>::digits == 113;
349 if (item_size == sizeof(long double) && long_double_is_float128) {
350 return "long double";
351 }
352 }
353 return external_type_name;
354 }

375 template<class T>
376 void SetValue(std::string name, std::vector<T> src) {
377 int item_size, total_bytes;
378 try {
379 item_size = GetVarItemsize(name);
380 total_bytes = GetVarNbytes(name);
381 }
382 catch (std::runtime_error &e) {
383 throw std::runtime_error("Cannot set " + name + " variable of " + model_name +
384 "; unable to test item and array sizes are equal (does variable " + name +
385 " exist for model " + model_name + "?)");
386 }
387 if (item_size != sizeof(src[0])) {
388 throw std::runtime_error("Cannot set " + name + " variable of " + model_name +
389 " with values of different item size");
390 }
391 if (src.size() != total_bytes / item_size) {
392 throw std::runtime_error(
393 "Cannot set " + name + " variable of " + model_name + " from vector of size " +
394 std::to_string(src.size()) + " (expected size " + std::to_string(total_bytes / item_size) +
395 ")");
396 }
397 SetValue(std::move(name), static_cast<void *>(src.data()));
398 }

635 std::shared_ptr<std::vector<std::string>> inner_get_variable_names(bool is_input_variables) {
636 // Writing this def once here at beginning: fewer lines, but this may or may not be used.
637 std::string varType = (is_input_variables) ? "input" : "output";
638
639 // Obtain this via inner functions, which should use the model directly and not other member functions
640 int variableCount;
641 try {
642 variableCount = (is_input_variables) ? inner_get_input_item_count() : inner_get_output_item_count();
643 }
644 catch (const std::exception &e) {
645 throw std::runtime_error(model_name + " failed to count of " + varType + " variable names array.");
646 }
647
648 // With variable count now obtained, create the vector
649 std::shared_ptr<std::vector<std::string>> var_names = std::make_shared<std::vector<std::string>>(
650 std::vector<std::string>(variableCount));
651
652 // Must get the names from the model as an array of C strings
653 // The array can be on the stack ...
654 std::vector<char*> names_array(variableCount);
655 // ... but allocate the space for the individual C strings (i.e., the char * elements)
656 for (int i = 0; i < variableCount; i++) {
657 names_array[i] = static_cast<char *>(malloc(sizeof(char) * BMI_MAX_VAR_NAME));
658 }
659
660 // With the necessary char** in hand, get the names from the model
661 int names_result;
662 if (is_input_variables) {
663 names_result = get_input_var_names(bmi_model.get(), names_array.data());
664 }
665 else {
666 names_result = get_output_var_names(bmi_model.get(), names_array.data());
667 }
668 if (names_result != BMI_SUCCESS) {
669 throw std::runtime_error(model_name + " failed to get array of " + varType + " variables names.");
670 }
671
672 // Then convert from array of C strings to vector of strings, freeing the allocated space as we go
673 for (int i = 0; i < variableCount; ++i) {
674 (*var_names)[i] = std::string(names_array[i]);
675 free(names_array[i]);
676 }
677
678 return var_names;
679 }
