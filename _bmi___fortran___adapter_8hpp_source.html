302 void SetValue(std::string name, std::vector<T> src) {
303 int item_size, total_bytes;
304 try {
305 item_size = GetVarItemsize(name);
306 total_bytes = GetVarNbytes(name);
307 }
308 catch (std::runtime_error &e) {
309 throw std::runtime_error("Cannot set " + name + " variable of " + model_name +

312 }
313 if (item_size != sizeof(src[0])) {
314 throw std::runtime_error("Cannot set " + name + " variable of " + model_name +
315 " with values of different item size");
316 }
317 if (src.size() != total_bytes / item_size) {
318 throw std::runtime_error(
319 "Cannot set " + name + " variable of " + model_name + " from vector of size " +
320 std::to_string(src.size()) + " (expected size " + std::to_string(total_bytes / item_size) +
321 ")");
322 }
323 SetValue(std::move(name), static_cast<void *>(src.data()));

597 std::shared_ptr<std::vector<std::string>> inner_get_variable_names(bool is_input_variables) {
598 // Writing this def once here at beginning: fewer lines, but this may or may not be used.
599 std::string varType = (is_input_variables) ? "input" : "output";
600
601 // Obtain this via inner functions, which should use the model directly and not other member functions
602 int variableCount;
603 try {
604 variableCount = (is_input_variables) ? inner_get_input_item_count() : inner_get_output_item_count();
605 }
606 catch (const std::exception &e) {
607 throw std::runtime_error(model_name + " failed to count of " + varType + " variable names array.");
608 }
609
610 // With variable count now obtained, create the vector
611 std::shared_ptr<std::vector<std::string>> var_names = std::make_shared<std::vector<std::string>>(
612 std::vector<std::string>(variableCount));
613
614 // Must get the names from the model as an array of C strings
615 // The array can be on the stack ...
616 std::vector<char*> names_array(variableCount);
617 // ... but allocate the space for the individual C strings (i.e., the char * elements)
618 for (int i = 0; i < variableCount; i++) {
619 names_array[i] = static_cast<char *>(malloc(sizeof(char) * BMI_MAX_VAR_NAME));
620 }
621
622 // With the necessary char** in hand, get the names from the model
623 int names_result;
624 if (is_input_variables) {
625 names_result = get_input_var_names(bmi_model.get(), names_array.data());
626 }
627 else {
628 names_result = get_output_var_names(bmi_model.get(), names_array.data());
629 }
630 if (names_result != BMI_SUCCESS) {
631 throw std::runtime_error(model_name + " failed to get array of " + varType + " variables names.");
632 }
633
634 // Then convert from array of C strings to vector of strings, freeing the allocated space as we go
635 for (int i = 0; i < variableCount; ++i) {
636 (*var_names)[i] = std::string(names_array[i]);
637 free(names_array[i]);
638 }
639
640 return var_names;
641 }
