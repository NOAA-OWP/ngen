92 inline double area_sqkm(const geojson::Feature& feature,
93 AreaSource preferred = AreaSource::attribute){
94 // Try the preferred source
95 std::optional<double> area = preferred == AreaSource::attribute
96 ? area_from_attribute(feature)
97 : area_from_geometry(*feature->geometry());
98 // The preferred source was nullopt, try the other one, the ternary switches
99 if (!area) {
100 area = preferred == AreaSource::attribute ? area_from_geometry(*feature->geometry())
101 : area_from_attribute(feature);
102 }
103 if (!area) {
104 throw std::invalid_argument("Catchment '" + feature->get_id() + "' has no " + area_v2 +
105 " or " + area_v1 + " attribute and no divide to measure, so the "
106 "area it distributes a value over is unknown.");
107 }
108 return *area;
