17The value-type contract shared between record producers and
18record-storage backends.
19
20A `Record` (in the `ngen::serialization` namespace) is the
21engine-agnostic interchange unit between code that *produces* a
22piece of saved state (e.g. a BMI adapter capturing a model's
23bytes) and code that *stores* that state (e.g. a file backend
24that lays it out on disk, an HDF5 backend that stores it as a
25dataset + attributes, or any future storage).
26
27This header defines only the value type and a small parse helper.
28It does NOT prescribe a wire format — backends are free to encode
29the fields however suits their storage.
30*/

74inline int64_t parse_epoch_string(const std::string& s) {
75 try {
76 std::size_t consumed = 0;
77 const long long v = std::stoll(s, &consumed);
78 while (consumed < s.size()
79 && std::isspace(static_cast<unsigned char>(s[consumed]))) {
80 ++consumed;
81 }
82 if (consumed == s.size()) {
83 return static_cast<int64_t>(v);
84 }
85 } catch (const std::exception&) {
86 // std::out_of_range / std::invalid_argument — fall through
87 // to the sentinel.
88 }
89 return UNPARSEABLE_TIMESTAMP_SENTINEL;
90}
91
99inline int64_t parse_formatted_time(const std::string& s) {
100 std::tm tm = {};
101 const char* end = ::strptime(s.c_str(), TIMESTAMP_STRPTIME_FORMAT, &tm);
102 if (end != nullptr) {
103 while (*end != '\0' && std::isspace(static_cast<unsigned char>(*end))) {
104 ++end;
105 }
106 if (*end == '\0') {
107 return static_cast<int64_t>(::timegm(&tm));
108 }
109 }
110 return UNPARSEABLE_TIMESTAMP_SENTINEL;
111}
112
120inline int64_t parse_timestamp(const std::string& s) {
121 if (s.empty()) return UNPARSEABLE_TIMESTAMP_SENTINEL;
122 const int64_t epoch = parse_epoch_string(s);
123 if (epoch != UNPARSEABLE_TIMESTAMP_SENTINEL) return epoch;
124 return parse_formatted_time(s);
125}

146// Shared field set for the two public checkpoint-record types
147// (`Record` and `RecordView`, below). Implementation alignment
148// mechanism only — adding or changing a metadata field touches
149// one definition and both aliases inherit the change.
150template <typename PayloadT>
151struct RecordT {
152 std::string id;
153 int64_t time_step = 0;
154 int64_t simulation_timestamp = 0;
155 int64_t checkpoint_epoch = 0;
156 PayloadT payload{};
157
158 RecordT() = default;
159 RecordT(std::string id_,
160 int64_t time_step_,
161 int64_t simulation_timestamp_,
162 PayloadT payload_,
163 int64_t checkpoint_epoch_ = 0)
164 : id(std::move(id_))
165 , time_step(time_step_)
166 , simulation_timestamp(simulation_timestamp_)
167 , checkpoint_epoch(checkpoint_epoch_)
168 , payload(std::move(payload_)) {}
169};
