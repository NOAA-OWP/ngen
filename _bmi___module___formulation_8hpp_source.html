141 std::vector<utils::OutputField> get_output_fields() const override {
142 // The output variables and their header (output) names are 1:1 and positionally aligned
143 // (enforced at construction) -- variable i is written under header i. Build one field per
144 // variable: units come from the model for that variable; the output name is its header
145 // (which defaults to the variable name, or is a configured alias).
146 const std::vector<std::string> &variables = get_output_variable_names();
147 boost::span<const std::string> headers = get_output_header_field_names();
148 std::vector<utils::OutputField> fields;
149 fields.reserve(variables.size());
150 for (std::size_t i = 0; i < variables.size(); ++i) {
151 std::optional<std::string> units;
152 try {
153 units = get_bmi_model()->GetVarUnits(variables[i]);
154 } catch (...) {
155 units = std::nullopt;
156 }
157 fields.emplace_back(variables[i], headers[i], std::move(units));
158 }
159 return fields;
160 }
