148 std::vector<utils::OutputField> get_output_fields() const override {
149 // The output variables and their header (output) names are 1:1 and positionally aligned
150 // (enforced at construction) -- variable i is written under header i. Build one field per
151 // variable: units come from the model for that variable; the output name is its header
152 // (which defaults to the variable name, or is a configured alias).
153 const std::vector<std::string> &variables = get_output_variable_names();
154 boost::span<const std::string> headers = get_output_header_field_names();
155 std::vector<utils::OutputField> fields;
156 fields.reserve(variables.size());
157 for (std::size_t i = 0; i < variables.size(); ++i) {
158 std::optional<std::string> units;
159 try {
160 units = get_bmi_model()->GetVarUnits(variables[i]);
161 } catch (...) {
162 units = std::nullopt;
163 }
164 fields.emplace_back(variables[i], headers[i], std::move(units));
165 }
166 return fields;
167 }
