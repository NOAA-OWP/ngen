34 Bmi_Var_Details(std::string name, std::string alias, const int item_size, const int num_items, models::bmi::primitive_type type, std::string units)
35 : name(std::move(name)), mapped_alias(std::move(alias)), type(type), units(std::move(units)), item_size(item_size), num_items(num_items)
36 {
37 // The buffers built from these details are sized by item_size and addressed as type, so
38 // the two disagreeing is caught once here rather than on every set.
39 if (models::bmi::size_of(this->type) != static_cast<size_t>(item_size)) {
40 throw std::runtime_error("Cannot use variable '" + this->name + "': its type " +
41 models::bmi::to_string(this->type) + " is " +
42 std::to_string(models::bmi::size_of(this->type)) +
43 " bytes, but its items are " + std::to_string(item_size) + " bytes.");
44 }

152 std::vector<utils::OutputField> get_output_fields() const override {
153 // The output variables and their header (output) names are 1:1 and positionally aligned
154 // (enforced at construction) -- variable i is written under header i. Build one field per
155 // variable: units come from the model for that variable; the output name is its header
156 // (which defaults to the variable name, or is a configured alias).
157 const std::vector<std::string> &variables = get_output_variable_names();
158 std::span<const std::string> headers = get_output_header_field_names();
159 std::vector<utils::OutputField> fields;
160 fields.reserve(variables.size());
161 for (std::size_t i = 0; i < variables.size(); ++i) {
162 std::optional<std::string> units;
163 try {
164 units = get_bmi_model()->GetVarUnits(variables[i]);
165 } catch (...) {
166 units = std::nullopt;
167 }
168 fields.emplace_back(variables[i], headers[i], std::move(units));
169 }
170 return fields;
171 }
