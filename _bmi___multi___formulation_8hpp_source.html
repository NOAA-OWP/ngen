551 inline void init_deferred_associations() {
552 for (int d = 0; d < deferredProviders.size(); ++d) {
553 std::shared_ptr<data_access::OptionalWrappedDataProvider> &deferredProvider = deferredProviders[d];
554 // Skip doing anything for any deferred provider that already has its backing provider set
555 if (deferredProvider->isWrappedProviderSet())
556 continue;
557
558 // TODO: improve this later; since defaults can be used, it is technically possible to grab something
559 // valid when something more appropriate would later be available
560 // Iterate through available data providers and set association once a sufficient one is found
561 std::map<std::string, std::shared_ptr<data_access::GenericDataProvider>>::iterator avail_it;
562 for (avail_it = availableData.begin(); avail_it != availableData.end(); avail_it++) {
563 // If this satisfies everything this deferred provider needs to provide, and thus can be/was set ...
564 if (deferredProvider->setWrappedProvider(avail_it->second.get())) {
565 break;
566 }
567 }
568
569 // If none of the available data providers could be used for this deferred one, throw exception
570 if (!deferredProvider->isWrappedProviderSet()) {
571 // TODO: this probably needs to be some kind of custom configuration exception
572 std::string msg = "Multi BMI formulation cannot be created from config: cannot find available data "
573 "provider to satisfy set of deferred provisions for nested module at index "
574 + std::to_string(deferredProviderModuleIndices[d]) + ": {";
575 // There must always be at least 1; get manually to help with formatting
576 msg += deferredProvider->get_available_variable_names()[0];
577 // And here make sure to start at 1 instead of 0
578 for (int i = 1; i < deferredProvider->get_available_variable_names().size(); ++i)
579 msg += ", " + deferredProvider->get_available_variable_names()[i];
580 msg += "}";
581 throw realization::ConfigurationException(msg);
582 }
583 }

601 template<class T>
602 std::shared_ptr<T> init_nested_module(int mod_index, std::string identifier, geojson::PropertyMap properties) {
603 std::shared_ptr<data_access::GenericDataProvider> wfp = std::make_shared<data_access::WrappedDataProvider>(this);
604 std::shared_ptr<T> mod = std::make_shared<T>(identifier, wfp, output);
605
606 // Since this is a nested formulation, support usage of the '{{id}}' syntax for init config file paths.
607 Catchment_Formulation::config_pattern_substitution(properties, BMI_REALIZATION_CFG_PARAM_REQ__INIT_CONFIG,
608 "{{id}}", id);
609
610 // Call create_formulation to perform the rest of the typical initialization steps for the formulation.
611 mod->create_formulation(properties);
612
613 // Set this up for placing in the module_variable_maps member variable
614 std::shared_ptr<std::map<std::string, std::string>> var_aliases;
615 var_aliases = std::make_shared<std::map<std::string, std::string>>(std::map<std::string, std::string>());
616 for (const std::string &var_name : mod->get_bmi_input_variables()) {
617 std::string framework_alias = mod->get_config_mapped_variable_name(var_name);
618 (*var_aliases)[framework_alias] = var_name;
619 // If framework_name is not yet in collection from which we have available data sources ...
620 if (availableData.count(framework_alias) != 1) {
621 setup_nested_deferred_provider(var_name, framework_alias, mod, mod_index);
622 }
623 else {
624 mod->input_forcing_providers[var_name] = availableData[framework_alias];
625 mod->input_forcing_providers[framework_alias] = availableData[framework_alias];
626 }
627 }
628
629 // Also add the output variable aliases
630 for (const std::string &var_name : mod->get_bmi_output_variables()) {
631 std::string framework_alias = mod->get_config_mapped_variable_name(var_name);
632 (*var_aliases)[framework_alias] = var_name;
633 if (availableData.count(framework_alias) > 0) {
634 throw std::runtime_error(
635 "Multi BMI cannot be created with module " + mod->get_model_type_name() +
636 " with output variable " + framework_alias +
637 (var_name == framework_alias ? "" : " (an alias of BMI variable " + var_name + ")") +
638 " because a previous module is using this output variable name/alias.");
639 }
640 availableData[framework_alias] = mod;
641 }
642 module_variable_maps[mod_index] = var_aliases;
643 return mod;
644 }

673 template<class T>
674 void setup_nested_deferred_provider(const std::string &bmi_input_var_name,
675 const std::string &framework_output_name,
676 std::shared_ptr<T> mod,
677 int mod_index) {
678 // TODO: probably don't actually need bmi_input_var_name, and just can deal with framework_output_name
679 // Create deferred, optional provider for providing this
680 // Only include BMI variable name, as that's what'll be visible when associating to backing provider
681 // It's "deferred" in that we'll set the backing later.
682 // It's "optional" in that it waits to use backing provider, using the default some number of times
683 std::shared_ptr<data_access::OptionalWrappedDataProvider> provider;
684 // TODO: make sure only alias is needed
685 auto defs_it = default_output_values.find(framework_output_name);
686 if (defs_it != default_output_values.end()) {
687 // TODO: consider also reading wait count from config
688 provider = std::make_shared<data_access::OptionalWrappedDataProvider>(framework_output_name, defs_it->second, 1);
689 }
690 else {
691 provider = std::make_shared<data_access::OptionalWrappedDataProvider>(framework_output_name);
692 }
693
694 // Add deferred to collection and module index to collection
695 deferredProviders.push_back(provider);
696 deferredProviderModuleIndices.push_back(mod_index);
697 // Assign as provider within module
698 // TODO: per TODO at top, probably can replace bmi_input_var_name here with framework_output_name
699 mod->input_forcing_providers[bmi_input_var_name] = provider;
700 }
