564 inline void init_deferred_associations() {
565 for (int d = 0; d < deferredProviders.size(); ++d) {
566 std::shared_ptr<data_access::OptionalWrappedDataProvider> &deferredProvider = deferredProviders[d];
567 // Skip doing anything for any deferred provider that already has its backing provider set
568 if (deferredProvider->isWrappedProviderSet())
569 continue;
570
571 // TODO: improve this later; since defaults can be used, it is technically possible to grab something
572 // valid when something more appropriate would later be available
573 // Iterate through available data providers and set association once a sufficient one is found
574 std::map<std::string, std::shared_ptr<data_access::GenericDataProvider>>::iterator avail_it;
575 for (avail_it = availableData.begin(); avail_it != availableData.end(); avail_it++) {
576 // If this satisfies everything this deferred provider needs to provide, and thus can be/was set ...
577 if (deferredProvider->setWrappedProvider(avail_it->second.get())) {
578 break;
579 }
580 }
581
582 // If none of the available data providers could be used for this deferred one, throw exception
583 if (!deferredProvider->isWrappedProviderSet()) {
584 // TODO: this probably needs to be some kind of custom configuration exception
585 std::string msg = "Multi BMI formulation cannot be created from config: cannot find available data "
586 "provider to satisfy set of deferred provisions for nested module at index "
587 + std::to_string(deferredProviderModuleIndices[d]) + ": {";
588 // There must always be at least 1; get manually to help with formatting
589 msg += deferredProvider->get_available_variable_names()[0];
590 // And here make sure to start at 1 instead of 0
591 for (int i = 1; i < deferredProvider->get_available_variable_names().size(); ++i)
592 msg += ", " + deferredProvider->get_available_variable_names()[i];
593 msg += "}";
594 throw realization::ConfigurationException(msg);
595 }
596 }

614 template<class T>
615 std::shared_ptr<T> init_nested_module(int mod_index, std::string identifier, geojson::PropertyMap properties) {
616 std::shared_ptr<data_access::GenericDataProvider> wfp = std::make_shared<data_access::WrappedDataProvider>(this);
617 std::shared_ptr<T> mod = std::make_shared<T>(identifier, wfp);
618
619 // A module's locator names where it sits, not what it models, so give it the
620 // subject this formulation models. Set before create_formulation, like the compound
621 // id below, so the module is never configured against the wrong one.
622 mod->set_feature_id(get_feature_id());
623
624 // Since this is a nested formulation, support usage of the '{{id}}' syntax for init
625 // config file paths. The subject, not this formulation's locator: a config writes
626 // {{id}} meaning the catchment, and the two differ once this is itself nested.
627 realization::config::pattern_substitution(properties, BMI_REALIZATION_CFG_PARAM_REQ__INIT_CONFIG,
628 "{{id}}", get_feature_id());
629
630 // Inject a three-part compound identity on the submodule BEFORE
631 // create_formulation runs, so that compound_id() is fully qualified
632 // during submodule construction with the following structure:
633 // "<catchment.submodule_index>:<submodule-mtn>:<multi-mtn>"
634 // Peek the submodule's own model_type_name from the config map —
635 // we cannot wait for create_formulation() to populate it on
636 // the submodule because we need the compound in place first.
637 std::string submodule_mtn =
638 properties.at(BMI_REALIZATION_CFG_PARAM_REQ__MODEL_TYPE).as_string();
639 mod->set_compound_id(identifier + ":" + submodule_mtn + ":" + get_model_type_name());
640
641 // Call create_formulation to perform the rest of the typical initialization steps for the formulation.
642 mod->create_formulation(properties);
643
644 // Set this up for placing in the module_variable_maps member variable
645 std::shared_ptr<std::map<std::string, std::string>> var_aliases;
646 var_aliases = std::make_shared<std::map<std::string, std::string>>(std::map<std::string, std::string>());
647 for (const std::string &var_name : mod->get_bmi_input_variables()) {
648 std::string framework_alias = mod->get_config_mapped_variable_name(var_name);
649 (*var_aliases)[framework_alias] = var_name;
650 // If framework_name is not yet in collection from which we have available data sources ...
651 if (availableData.count(framework_alias) != 1) {
652 setup_nested_deferred_provider(var_name, framework_alias, mod, mod_index);
653 }
654 else {
655 mod->input_forcing_providers[var_name] = availableData[framework_alias];
656 mod->input_forcing_providers[framework_alias] = availableData[framework_alias];
657 }
658 }
659
660 // Also add the output variable aliases
661 for (const std::string &var_name : mod->get_bmi_output_variables()) {
662 std::string framework_alias = mod->get_config_mapped_variable_name(var_name);
663 (*var_aliases)[framework_alias] = var_name;
664 if (availableData.count(framework_alias) > 0) {
665 throw std::runtime_error(
666 "Multi BMI cannot be created with module " + mod->get_model_type_name() +
667 " with output variable " + framework_alias +
668 (var_name == framework_alias ? "" : " (an alias of BMI variable " + var_name + ")") +
669 " because a previous module is using this output variable name/alias.");
670 }
671 availableData[framework_alias] = mod;
672 }
673 module_variable_maps[mod_index] = var_aliases;
674 return mod;
675 }

704 template<class T>
705 void setup_nested_deferred_provider(const std::string &bmi_input_var_name,
706 const std::string &framework_output_name,
707 std::shared_ptr<T> mod,
708 int mod_index) {
709 // TODO: probably don't actually need bmi_input_var_name, and just can deal with framework_output_name
710 // Create deferred, optional provider for providing this
711 // Only include BMI variable name, as that's what'll be visible when associating to backing provider
712 // It's "deferred" in that we'll set the backing later.
713 // It's "optional" in that it waits to use backing provider, using the default some number of times
714 std::shared_ptr<data_access::OptionalWrappedDataProvider> provider;
715 // TODO: make sure only alias is needed
716 auto defs_it = default_output_values.find(framework_output_name);
717 if (defs_it != default_output_values.end()) {
718 // TODO: consider also reading wait count from config
719 provider = std::make_shared<data_access::OptionalWrappedDataProvider>(framework_output_name, defs_it->second, 1);
720 }
721 else {
722 provider = std::make_shared<data_access::OptionalWrappedDataProvider>(framework_output_name);
723 }
724
725 // Add deferred to collection and module index to collection
726 deferredProviders.push_back(provider);
727 deferredProviderModuleIndices.push_back(mod_index);
728 // Assign as provider within module
729 // TODO: per TODO at top, probably can replace bmi_input_var_name here with framework_output_name
730 mod->input_forcing_providers[bmi_input_var_name] = provider;
731 }
