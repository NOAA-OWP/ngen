14/*
15 * Typed access to BMI model variables.
16 *
17 * A BMI adapter reports a variable in three ways the standard does not tie together: a type, as a
18 * string; a size, as a count of bytes; and a buffer or pointer carrying no type at all. Reading or
19 * writing one therefore means choosing a C++ type, sizing a buffer, and trusting the two agree.
20 *
21 * Prefer the helpers below to calling Bmi_Adapter::GetValue() or GetValuePtr() and casting the
22 * result:
23 *
24 * with_bmi_type() resolve a type name to a C++ type; the one table the others are built on
25 * GetValue<T>() read a variable, converted to T
26 * GetValuePtr<T>() borrow the model's own storage, typed
27 * get_value_as_type() build a value in the variable's declared type
28 * get_values_as_type() build a buffer in the variable's declared type
29 * require_item_size() confirm a type name and a reported item size describe the same width
30 *
31 * The raw adapter calls stay public -- the adapters implement them, and a caller may genuinely want
32 * bytes -- but they are not the ordinary way to read or write a variable.
33 */

164 inline void require_item_size(const std::string &type, size_t item_size, const std::string &described)
165 {
166 const size_t declared = with_bmi_type(type, []<typename T>() -> size_t { return sizeof(T); });
167 if (declared != item_size) {
168 throw std::runtime_error("Cannot use " + described + ": type '" + type + "' is " +
169 std::to_string(declared) + " bytes, but the model reports an item size of " +
170 std::to_string(item_size) + " bytes.");
171 }

270 T *GetValuePtr(Bmi_Adapter &model, const std::string &name)
271 {
272 const size_t item_size = static_cast<size_t>(model.GetVarItemsize(name));
273 if (item_size != sizeof(T)) {
274 throw std::runtime_error("Cannot get pointer to variable " + name + " of model " +

276 " byte type; its items are " + std::to_string(item_size) + " bytes.");
277 }
278 return static_cast<T *>(model.GetValuePtr(name));

290 template <typename T>
291 std::vector<T> GetValue(Bmi_Adapter& model, const std::string& name) {
292 //TODO make model const ref
293 int total_mem = model.GetVarNbytes(name);
294 int item_size = model.GetVarItemsize(name);
295 if( total_mem == 0 || item_size == 0){
296 // Early stop if no items to be returned (also prevents possible divide by 0 computing num_items)
297 // Becuase this function is used by others which assume the return value to have AT LEAST
298 // 1 value, then this is a terminal error. This happens when the BMI model hasn't properly
299 // initialized/allocated/set the value to something meaningful. This can happen, for example,
300 // with grid data if the variables themselves are dynamically allocated based on grid properties
301 // but the grid is 0, then we essentially get a sentinal pointer with no data in it.
302 // one possible way to handle this would be to ensure a single element vector with NaN were returned here
303 // but since this function has to deal with both int and floating point types, that isn't possible
304 // because int has not NaN representation. So the best we can do at this point is raise a runtime exception
305 throw std::runtime_error("Unable to get value of variable "+name+". Model "+ model.get_model_name() +
306 " reports no valid items (Nbytes = "+std::to_string(total_mem) +
307 ", Itemsize = "+std::to_string(item_size)+".");
308 }
309 int num_items = total_mem/item_size;
310 //Determine what type we need to cast from
311 std::string type = model.get_analogous_cxx_type(model.GetVarType(name), item_size);
312
313 //C++ form of malloc
314 void* data = ::operator new(total_mem); //Possible to allocate 0 bytes...
315 // Use smart pointer to ensure cleanup on throw/out of scope...
316 // This works, and is relatively cheap since the lambda is stateless, only one instance should be created.
317 auto sptr = std::shared_ptr<void>(data, [](void *p) { ::operator delete(p); });
318 //Delegate to specific adapter's GetValue()
319 //Note, may be able to optimize this furthur using GetValuePtr
320 //which would avoid copying in the BMI model and copying again here
321 model.GetValue(name, data);
322
323 try {
324 require_item_size(type, static_cast<size_t>(item_size),
325 "variable '" + name + "' of model '" + model.get_model_name() + "'");
326 return with_bmi_type(type, [&]<typename S>() -> std::vector<T> {
327 return helper::make_vector<T>(static_cast<const S*>(data), num_items);
328 });
329 } catch (const unknown_bmi_type&) {
330 // The shared table knows the type name, but not what was being read or as what.
331 throw std::runtime_error("Unable to get value of variable " + name +
332 " as " + boost::typeindex::type_id<T>().pretty_name() + ": no logic for converting variable type " + type);
333 }
334 }
335 }
