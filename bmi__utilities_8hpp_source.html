17/*
18 * Typed access to BMI model variables.
19 *
20 * A BMI adapter reports a variable in three ways the standard does not tie together: a type, as a
21 * string; a size, as a count of bytes; and a buffer or pointer carrying no type at all. Reading or
22 * writing one therefore means choosing a C++ type, sizing a buffer, and trusting the two agree.
23 *
24 * A type name is resolved once, to a primitive_type (see bmi_primitive_type.hpp), and that is what
25 * is passed around after.
26 *
27 * Prefer the helpers below to calling Bmi_Adapter::GetValue() or GetValuePtr() and casting the
28 * result:
29 *
30 * GetValue<T>() read a variable, or one item of it, converted to T
31 * GetValuePtr<T>() borrow the model's own storage, typed
32 * get_value_as_type() build a value in the variable's declared type
33 * get_values_as_type() build a buffer in the variable's declared type
34 * resolve_var_type() the primitive_type to address a variable as, checked against its item size
35 *
36 * The raw adapter calls stay public -- the adapters implement them, and a caller may genuinely want
37 * bytes -- but they are not the ordinary way to read or write a variable.
38 */

55 inline primitive_type resolve_var_type(Bmi_Adapter &model, std::string_view name)
56 {
57 // The adapter calls take the name as a std::string, by value, so make one and hand it on.
58 std::string var_name(name);
59 const size_t item_size = static_cast<size_t>(model.GetVarItemsize(var_name));
60 const std::string reported = model.GetVarType(std::move(var_name));
61 const primitive_type type = model.get_primitive_type(reported, item_size);
62 // Every buffer is sized from the item size and addressed as the type, so a disagreement reads or
63 // writes past the end -- undetectable afterwards, so refuse up front.
64 if (size_of(type) != item_size) {
65 throw std::runtime_error("Cannot use variable '" + std::string(name) + "' of model '" +
66 model.get_model_name() + "': its type '" + reported + "' is " +
67 std::to_string(size_of(type)) +
68 " bytes, but the model reports an item size of " +
69 std::to_string(item_size) + " bytes.");
70 }
71 return type;
72 }

170 T *GetValuePtr(Bmi_Adapter &model, const std::string &name)
171 {
172 const size_t item_size = static_cast<size_t>(model.GetVarItemsize(name));
173 if (item_size != sizeof(T)) {
174 throw std::runtime_error("Cannot get pointer to variable " + name + " of model " +
175 model.get_model_name() + " as a " + std::to_string(sizeof(T)) +
176 " byte type; its items are " + std::to_string(item_size) + " bytes.");
177 }
178 return static_cast<T *>(model.GetValuePtr(name));

190 template <typename T>
191 std::vector<T> GetValue(Bmi_Adapter& model, const std::string& name) {
192 //TODO make model const ref
193 int total_mem = model.GetVarNbytes(name);
194 int item_size = model.GetVarItemsize(name);
195 if( total_mem == 0 || item_size == 0){
196 // Early stop if no items to be returned (also prevents possible divide by 0 computing num_items)
197 // Becuase this function is used by others which assume the return value to have AT LEAST
198 // 1 value, then this is a terminal error. This happens when the BMI model hasn't properly
199 // initialized/allocated/set the value to something meaningful. This can happen, for example,
200 // with grid data if the variables themselves are dynamically allocated based on grid properties
201 // but the grid is 0, then we essentially get a sentinal pointer with no data in it.
202 // one possible way to handle this would be to ensure a single element vector with NaN were returned here
203 // but since this function has to deal with both int and floating point types, that isn't possible
204 // because int has not NaN representation. So the best we can do at this point is raise a runtime exception
205 throw std::runtime_error("Unable to get value of variable "+name+". Model "+ model.get_model_name() +
206 " reports no valid items (Nbytes = "+std::to_string(total_mem) +
207 ", Itemsize = "+std::to_string(item_size)+".");
208 }
209 int num_items = total_mem/item_size;
210 //Determine what type we need to cast from, before the model writes into a buffer sized by it
211 primitive_type type;
212 try {
213 type = resolve_var_type(model, name);
214 } catch (const unknown_bmi_type& e) {
215 // The shared table knows the type name, but not what was being read or as what.
216 throw std::runtime_error("Unable to get value of variable " + name + " of model " +
217 model.get_model_name() + " as " +
218 boost::typeindex::type_id<T>().pretty_name() +
219 ": no logic for converting variable type " + e.type_name);
220 }
221
222 //C++ form of malloc
223 void* data = ::operator new(total_mem); //Possible to allocate 0 bytes...
224 // Use smart pointer to ensure cleanup on throw/out of scope...
225 // This works, and is relatively cheap since the lambda is stateless, only one instance should be created.
226 auto sptr = std::shared_ptr<void>(data, [](void *p) { ::operator delete(p); });
227 //Delegate to specific adapter's GetValue()
228 //Note, may be able to optimize this furthur using GetValuePtr
229 //which would avoid copying in the BMI model and copying again here
230 model.GetValue(name, data);
231
232 return with_primitive_type(type, [&]<typename S>() -> std::vector<T> {
233 return helper::make_vector<T>(static_cast<const S*>(data), num_items);
234 });
235 }

251 template <typename T>
252 T GetValue(Bmi_Adapter& model, const std::string& name, int index) {
253 const int item_size = model.GetVarItemsize(name);
254 const int num_items = item_size > 0 ? model.GetVarNbytes(name) / item_size : 0;
255 if (index < 0 || index >= num_items) {
256 throw std::out_of_range("Cannot get item " + std::to_string(index) + " of variable " + name +
257 " of model " + model.get_model_name() + ", which has " +
258 std::to_string(num_items) + " items.");
259 }
260
261 primitive_type type;
262 try {
263 type = resolve_var_type(model, name);
264 } catch (const unknown_bmi_type& e) {
265 throw std::runtime_error("Unable to get value of variable " + name + " of model " +
266 model.get_model_name() + " as " +
267 boost::typeindex::type_id<T>().pretty_name() +
268 ": no logic for converting variable type " + e.type_name);
269 }
270
271 return with_primitive_type(type, [&]<typename S>() -> T {
272 S item;
273 model.GetValueAtIndices(name, &item, &index, 1);
274 return static_cast<T>(item);
275 });
