14/*
15 * Typed access to BMI model variables.
16 *
17 * A BMI adapter reports a variable in three ways the standard does not tie together: a type, as a
18 * string; a size, as a count of bytes; and a buffer or pointer carrying no type at all. Reading or
19 * writing one therefore means choosing a C++ type, sizing a buffer, and trusting the two agree.
20 *
21 * Prefer the helpers below to calling Bmi_Adapter::GetValue() or GetValuePtr() and casting the
22 * result:
23 *
24 * with_bmi_type() resolve a type name to a C++ type; the one table the others are built on
25 * GetValue<T>() read a variable, or one item of it, converted to T
26 * GetValuePtr<T>() borrow the model's own storage, typed
27 * get_value_as_type() build a value in the variable's declared type
28 * get_values_as_type() build a buffer in the variable's declared type
29 * require_item_size() confirm a type name and a reported item size describe the same width
30 * resolve_var_type() the type name to address a model's variable as, checked against its item size
31 *
32 * The raw adapter calls stay public -- the adapters implement them, and a caller may genuinely want
33 * bytes -- but they are not the ordinary way to read or write a variable.
34 */

165 inline void require_item_size(const std::string &type, size_t item_size, const std::string &described)
166 {
167 const size_t declared = with_bmi_type(type, []<typename T>() -> size_t { return sizeof(T); });
168 if (declared != item_size) {
169 throw std::runtime_error("Cannot use " + described + ": type '" + type + "' is " +
170 std::to_string(declared) + " bytes, but the model reports an item size of " +
171 std::to_string(item_size) + " bytes.");
172 }

187 inline std::string resolve_var_type(Bmi_Adapter &model, const std::string &name)
188 {
189 const size_t item_size = static_cast<size_t>(model.GetVarItemsize(name));
190 std::string type = model.get_analogous_cxx_type(model.GetVarType(name), item_size);
191 require_item_size(type, item_size, "variable '" + name + "' of model '" + model.get_model_name() + "'");
192 return type;
193 }

291 T *GetValuePtr(Bmi_Adapter &model, const std::string &name)
292 {
293 const size_t item_size = static_cast<size_t>(model.GetVarItemsize(name));
294 if (item_size != sizeof(T)) {
295 throw std::runtime_error("Cannot get pointer to variable " + name + " of model " +

297 " byte type; its items are " + std::to_string(item_size) + " bytes.");
298 }
299 return static_cast<T *>(model.GetValuePtr(name));

311 template <typename T>
312 std::vector<T> GetValue(Bmi_Adapter& model, const std::string& name) {
313 //TODO make model const ref
314 int total_mem = model.GetVarNbytes(name);
315 int item_size = model.GetVarItemsize(name);
316 if( total_mem == 0 || item_size == 0){
317 // Early stop if no items to be returned (also prevents possible divide by 0 computing num_items)
318 // Becuase this function is used by others which assume the return value to have AT LEAST
319 // 1 value, then this is a terminal error. This happens when the BMI model hasn't properly
320 // initialized/allocated/set the value to something meaningful. This can happen, for example,
321 // with grid data if the variables themselves are dynamically allocated based on grid properties
322 // but the grid is 0, then we essentially get a sentinal pointer with no data in it.
323 // one possible way to handle this would be to ensure a single element vector with NaN were returned here
324 // but since this function has to deal with both int and floating point types, that isn't possible
325 // because int has not NaN representation. So the best we can do at this point is raise a runtime exception
326 throw std::runtime_error("Unable to get value of variable "+name+". Model "+ model.get_model_name() +
327 " reports no valid items (Nbytes = "+std::to_string(total_mem) +
328 ", Itemsize = "+std::to_string(item_size)+".");
329 }
330 int num_items = total_mem/item_size;
331 //Determine what type we need to cast from, before the model writes into a buffer sized by it
332 std::string type;
333 try {
334 type = resolve_var_type(model, name);
335 } catch (const unknown_bmi_type& e) {
336 // The shared table knows the type name, but not what was being read or as what.
337 throw std::runtime_error("Unable to get value of variable " + name + " of model " +
338 model.get_model_name() + " as " +
339 boost::typeindex::type_id<T>().pretty_name() +
340 ": no logic for converting variable type " + e.type_name);
341 }
342
343 //C++ form of malloc
344 void* data = ::operator new(total_mem); //Possible to allocate 0 bytes...
345 // Use smart pointer to ensure cleanup on throw/out of scope...
346 // This works, and is relatively cheap since the lambda is stateless, only one instance should be created.
347 auto sptr = std::shared_ptr<void>(data, [](void *p) { ::operator delete(p); });
348 //Delegate to specific adapter's GetValue()
349 //Note, may be able to optimize this furthur using GetValuePtr
350 //which would avoid copying in the BMI model and copying again here
351 model.GetValue(name, data);
352
353 return with_bmi_type(type, [&]<typename S>() -> std::vector<T> {
354 return helper::make_vector<T>(static_cast<const S*>(data), num_items);
355 });
356 }

372 template <typename T>
373 T GetValue(Bmi_Adapter& model, const std::string& name, int index) {
374 const int item_size = model.GetVarItemsize(name);
375 const int num_items = item_size > 0 ? model.GetVarNbytes(name) / item_size : 0;
376 if (index < 0 || index >= num_items) {
377 throw std::out_of_range("Cannot get item " + std::to_string(index) + " of variable " + name +
378 " of model " + model.get_model_name() + ", which has " +
379 std::to_string(num_items) + " items.");
380 }
381
382 std::string type;
383 try {
384 type = resolve_var_type(model, name);
385 } catch (const unknown_bmi_type& e) {
386 throw std::runtime_error("Unable to get value of variable " + name + " of model " +
387 model.get_model_name() + " as " +
388 boost::typeindex::type_id<T>().pretty_name() +
389 ": no logic for converting variable type " + e.type_name);
390 }
391
392 return with_bmi_type(type, [&]<typename S>() -> T {
393 S item;
394 model.GetValueAtIndices(name, &item, &index, 1);
395 return static_cast<T>(item);
396 });
397 }
