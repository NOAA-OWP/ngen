18Shared on-disk record format for ngen BMI state serialization.
19
20This header is the public API for producing and consuming
21serialization records. The on-disk layout is the responsibility of
22the internal `wire_format.hpp`, which lives under
23`src/utilities/bmi/` and is not exported. External code calls into
24this header's functions:
25
26 write_record(out, rec) — append a record
27 read_next_record(in, rec) — read the next record
28 read_record_length(in, body_out) — read prefix; report body length
29 read_record_metadata(in, prefix, id) — read prefix + id; skip payload
30
31The value type `SerializationRecord` itself lives in the engine-
32agnostic `ngen::serialization` library (see
33`include/utilities/serialization/record.hpp`) and is
34re-exported under `models::bmi::protocols` for source compatibility
35with existing call sites. Future storage backends share the same
36value type without depending on the BMI library.
37
38On-disk layout
39--------------
40A checkpoint file is a concatenation of records. Each record is:
41
42 [ fixed-size prefix ][ id_length bytes of id ][ payload_length bytes of payload ]
43
44The prefix layout is defined in `wire_format.hpp` and includes the
45magic bytes "NGSR" (which identify the file format), a wire-format
46version, simulation step / timestamp / wall-clock-epoch metadata,
47and the two body-length fields. Anyone with a hex editor can
48identify a checkpoint file by its "NGSR..." prefix.
49
50See `wire_format.hpp` for the rationale behind the chosen
51byte-level layout (cross-host portability, library independence,
52index-build performance).
53
54Truncation tolerance
55--------------------
56A torn final record (writer crashed mid-write, leaving fewer than
57the expected bytes on disk) is detected and treated as
58end-of-stream rather than as a corrupt file. The read functions
59return `false` on a short read at any field boundary — the prefix,
60the id bytes, or the payload bytes — which is what
61restart-from-crash workflows depend on.
62
63Uniqueness
64----------
65Each `(id, time_step)` pair is unique within a file. One entity may
66write multiple records at different time steps, but only one
67record per step.
68*/

123#define NGEN_BMI_MAX_RECORD_PAYLOAD_BYTES (1ULL << 30) // 1 GiB
124#endif
125constexpr uint64_t MAX_RECORD_PAYLOAD_BYTES = NGEN_BMI_MAX_RECORD_PAYLOAD_BYTES;
126
127// Internal helper shared by read_next_record, read_record_length,
128// and read_record_metadata. Reads the prefix and validates the
129// magic, the wire_version, and the payload_length cap. Returns
130// `Status::Ok` on a fully-read valid prefix, `Status::Eof` on a
131// clean short-read (EOF or torn prefix), or an error-arm string
132// on any validation failure of a fully-read prefix. Internal —
133// callers in this header always check the return value, so no
134// nodiscard.
135inline auto
136read_validated_prefix(std::istream& in, serialization::wire_format::RecordPrefix& prefix)
137 -> expected<Status, std::string> {
138 if (const auto s = serialization::wire_format::read_record_prefix(in, prefix);
139 s == Status::Eof) {
140 return s; // EOF or torn prefix — clean stop, propagate
141 }
142 if (prefix.magic != serialization::wire_format::RecordPrefix::MAGIC) {
143 return make_unexpected(
144 "serialization_record: bad magic 0x" + std::to_string(prefix.magic)
145 + " — file is not a v0.2 record stream or is corrupted "
146 "at the record boundary"
147 );
148 }
149 if (prefix.wire_version != serialization::wire_format::RecordPrefix::VERSION) {
150 return make_unexpected(
151 "serialization_record: unsupported wire_version " + std::to_string(prefix.wire_version)
152 + " (this reader supports v"
153 + std::to_string(serialization::wire_format::RecordPrefix::VERSION) + ")"
154 );
155 }
156 if (prefix.payload_length > MAX_RECORD_PAYLOAD_BYTES) {
157 return make_unexpected(
158 "serialization_record: payload_length " + std::to_string(prefix.payload_length)
159 + " exceeds MAX_RECORD_PAYLOAD_BYTES sanity cap"
160 );
161 }
162 return Status::Ok;
163}

185nsel_NODISCARD inline auto write_record(std::ostream& out, const SerializationRecordView& rec)
186 -> expected<void, std::string> {
187 serialization::wire_format::RecordPrefix prefix;
188 prefix.time_step = rec.time_step;
189 prefix.simulation_timestamp = rec.simulation_timestamp;
190 prefix.checkpoint_epoch = rec.checkpoint_epoch;
191 prefix.id_length = static_cast<uint16_t>(rec.id.size());
192 prefix.payload_length = static_cast<uint64_t>(rec.payload.size());
193 // (magic and wire_version come from the struct's default values.)
194
195 serialization::wire_format::write_record_prefix(out, prefix);
196 if (!rec.id.empty()) {
197 out.write(rec.id.data(), static_cast<std::streamsize>(rec.id.size()));
198 }
199 if (!rec.payload.empty()) {
200 out.write(rec.payload.data(), static_cast<std::streamsize>(rec.payload.size()));
201 }
202 if (!out) {
203 return make_unexpected(
204 std::string("serialization_record: stream failed during record write")
205 );
206 }
207 return {};
208}

256nsel_NODISCARD inline auto read_next_record(std::istream& in, SerializationRecord& rec)
257 -> expected<Status, std::string> {
258 serialization::wire_format::RecordPrefix prefix;
259 auto p = read_validated_prefix(in, prefix);
260 // Propagate any non-Ok outcome unchanged — error arm or Eof.
261 if (!p || p.value() == Status::Eof) return p;
262
263 // id body
264 std::string id;
265 id.resize(prefix.id_length);
266 if (prefix.id_length > 0) {
267 in.read(&id[0], static_cast<std::streamsize>(prefix.id_length));
268 if (static_cast<uint64_t>(in.gcount()) != prefix.id_length) {
269 return Status::Eof; // torn id — clean stop
270 }
271 }
272
273 // payload body
274 std::vector<char> payload;
275 payload.resize(static_cast<size_t>(prefix.payload_length));
276 if (prefix.payload_length > 0) {
277 in.read(payload.data(), static_cast<std::streamsize>(prefix.payload_length));
278 if (static_cast<uint64_t>(in.gcount()) != prefix.payload_length) {
279 return Status::Eof; // torn payload — clean stop
280 }
281 }
282
283 rec.id = std::move(id);
284 rec.time_step = prefix.time_step;
285 rec.simulation_timestamp = prefix.simulation_timestamp;
286 rec.checkpoint_epoch = prefix.checkpoint_epoch;
287 rec.payload = std::move(payload);
288 return Status::Ok;
289}

308nsel_NODISCARD inline auto read_record_length(std::istream& in, uint64_t& body_out)
309 -> expected<Status, std::string> {
310 serialization::wire_format::RecordPrefix prefix;
311 auto p = read_validated_prefix(in, prefix);
312 // Propagate any non-Ok outcome unchanged — error arm or Eof.
313 if (!p || p.value() == Status::Eof) return p;
314 body_out = static_cast<uint64_t>(prefix.id_length) + prefix.payload_length;
315 return Status::Ok;
316}

333nsel_NODISCARD inline auto read_record_metadata(
334 std::istream& in,
335 serialization::wire_format::RecordPrefix& prefix_out,
336 std::string& id_out
337) -> expected<Status, std::string> {
338 auto p = read_validated_prefix(in, prefix_out);
339 // Propagate any non-Ok outcome unchanged — error arm or Eof.
340 if (!p || p.value() == Status::Eof) return p;
341
342 id_out.resize(prefix_out.id_length);
343 if (prefix_out.id_length > 0) {
344 in.read(&id_out[0], static_cast<std::streamsize>(prefix_out.id_length));
345 if (static_cast<uint64_t>(in.gcount()) != prefix_out.id_length) {
346 return Status::Eof; // torn id — clean stop
347 }
348 }
349
350 if (prefix_out.payload_length > 0) {
351 in.seekg(static_cast<std::streamoff>(prefix_out.payload_length), std::ios::cur);
352 if (!in) return Status::Eof; // stream went bad while seeking
353 }
354 return Status::Ok;
355}
