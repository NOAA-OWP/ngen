65inline void apply_serialization_restore_subset_default(
66 geojson::PropertyMap& global_configs,
67 const std::vector<std::string>& known_ids)
68{
69 if (known_ids.empty()) return;
70
71 const char* ser_key = to_key_string(GlobalConfigKey::SERIALIZATION);
72 auto ser_it = global_configs.find(ser_key);
73 if (ser_it == global_configs.end()) return;
74 if (ser_it->second.get_type() != geojson::PropertyType::Object) return;
75
76 geojson::PropertyMap ser_map = ser_it->second.get_values();
77
78 auto restore_it = ser_map.find("restore");
79 if (restore_it == ser_map.end()) return;
80 if (restore_it->second.get_type() != geojson::PropertyType::Object) return;
81
82 geojson::PropertyMap restore_map = restore_it->second.get_values();
83 if (restore_map.count("id_subset") != 0) return; // caller already set it
84
85 // Build the id_subset list as a List-typed JSONProperty whose
86 // elements are string-typed JSONProperties (matching how the
87 // ptree-based constructor emits JSON arrays of strings).
88 std::vector<geojson::JSONProperty> id_list;
89 id_list.reserve(known_ids.size());
90 for (const auto& id : known_ids) {
91 id_list.emplace_back(std::string(""), id);
92 }
93 restore_map.emplace(
94 "id_subset",
95 geojson::JSONProperty(std::string("id_subset"), std::move(id_list))
96 );
97
98 // Rebuild the nested objects with the new map. JSONProperty's
99 // PropertyMap-ctor takes a non-const reference and captures the
100 // map by value into its internal field, so these locals are the
101 // actual owners of the reconstituted tree.
102 geojson::JSONProperty restore_prop(std::string("restore"), restore_map);
103 ser_map.erase("restore");
104 ser_map.emplace("restore", std::move(restore_prop));
105
106 geojson::JSONProperty ser_prop(std::string(ser_key), ser_map);
107 global_configs.erase(ser_key);
108 global_configs.emplace(ser_key, std::move(ser_prop));
109}
