104 template <typename T>
105 constexpr primitive_type primitive_type_of()
106 {
107 if constexpr (std::is_same_v<T, long double>) {
108 return primitive_type::long_double;
109 } else if constexpr (std::is_same_v<T, double>) {
110 return primitive_type::f64;
111 } else if constexpr (std::is_same_v<T, float>) {
112 return primitive_type::f32;
113 } else {
114 static_assert(std::is_integral_v<T> && !std::is_same_v<T, bool>, "No primitive_type for this type");
115 static_assert(sizeof(T) == 1 || sizeof(T) == 2 || sizeof(T) == 4 || sizeof(T) == 8,
116 "No primitive_type of this integer's width");
117 constexpr bool is_signed = std::is_signed_v<T>;
118 if constexpr (sizeof(T) == 1) return is_signed ? primitive_type::i8 : primitive_type::u8;
119 else if constexpr (sizeof(T) == 2) return is_signed ? primitive_type::i16 : primitive_type::u16;
120 else if constexpr (sizeof(T) == 4) return is_signed ? primitive_type::i32 : primitive_type::u32;
121 else return is_signed ? primitive_type::i64 : primitive_type::u64;
122 }
123 }

171 template <typename F>
172 auto with_primitive_type(primitive_type type, F &&f)
173 {
174 switch (type) {
175 case primitive_type::i8: return f.template operator()<std::int8_t>();
176 case primitive_type::u8: return f.template operator()<std::uint8_t>();
177 case primitive_type::i16: return f.template operator()<std::int16_t>();
178 case primitive_type::u16: return f.template operator()<std::uint16_t>();
179 case primitive_type::i32: return f.template operator()<std::int32_t>();
180 case primitive_type::u32: return f.template operator()<std::uint32_t>();
181 case primitive_type::i64: return f.template operator()<std::int64_t>();
182 case primitive_type::u64: return f.template operator()<std::uint64_t>();
183 case primitive_type::f32: return f.template operator()<float>();
184 case primitive_type::f64: return f.template operator()<double>();
185 case primitive_type::long_double: return f.template operator()<long double>();
186 case primitive_type::f128:
187 // Where long double is not binary128, GCC and Clang offer __float128 on x86-64, and C++23
188 // std::float128_t where <stdfloat> defines it. Supporting f128 there means dispatching to
189 // that type here, which every callable then has to compile for, as for long double.
190 if constexpr (long_double_is_float128) {
191 return f.template operator()<long double>();
192 } else {
193 break;
194 }
195 }
196 throw unsupported_primitive_type(type);
197 }
