17Abstract record-storage interface.
18
19`RecordBackend` is the engine-agnostic abstraction over "how
20does this `Record` end up persisted and retrievable later."
21Concrete backends implement the interface and choose their own
22storage details; the abstract surface is silent on byte layout.
23
24This file declares the abstract surface and per-method API
25contracts. Cross-cutting design — sub-handle lifetime patterns,
26error categorization, single-in-flight rule, durability,
27identity, recommended call patterns — lives in this directory's
28README.md; that's the discoverable reference for backend
29implementers.
30*/

70struct BackendError {
71 enum class Kind { NotFound, Corrupted, IOError };
72 Kind kind;
73 std::string message;
74
75 BackendError() = default;
76 BackendError(Kind k, std::string m)
77 : kind(k), message(std::move(m)) {}
78};

83class RecordBackend {
84public:
88 class Writer {
89 public:
90 virtual ~Writer() = default;
91
98 [[nodiscard]] virtual auto write(const RecordView& rv)
99 -> expected<void, BackendError> = 0;
100
107 [[nodiscard]] virtual auto commit()
108 -> expected<void, BackendError> = 0;
109 };

117 class Reader {
118 public:
119 virtual ~Reader() = default;
120
124 [[nodiscard]] virtual auto find_latest(const std::string& id)
125 -> expected<Record, BackendError> = 0;
126
130 [[nodiscard]] virtual auto find_at_step(const std::string& id,
131 int64_t time_step)
132 -> expected<Record, BackendError> = 0;
133
138 [[nodiscard]] virtual auto

140 int64_t simulation_timestamp)
141 -> expected<Record, BackendError> = 0;
142
148 [[nodiscard]] virtual auto close() -> expected<void, BackendError> {
149 return {};
150 }
151 };
152
153 virtual ~RecordBackend() = default;
154
160 [[nodiscard]] virtual auto
161 writer(std::chrono::system_clock::time_point epoch,
162 Durability durability)
163 -> expected<std::unique_ptr<Writer>, BackendError> = 0;
164
170 [[nodiscard]] virtual auto reader(IdPredicate in_scope = {})
171 -> expected<std::unique_ptr<Reader>, BackendError> = 0;
172
197 [[nodiscard]] virtual auto finalize()
198 -> expected<void, std::vector<BackendError>> = 0;
199
200 // Scoped convenience wrappers — recommended call pattern.
201 // Non-virtual; concrete backends do not override them. See
202 // the README for why they exist and what they enforce.
203
208 template <typename F>
209 [[nodiscard]] auto with_writer(std::chrono::system_clock::time_point epoch,
210 Durability durability, F&& fn)
211 -> expected<void, BackendError>
212 {
213 auto wo = writer(epoch, durability);
214 if (!wo) return make_unexpected(std::move(wo.error()));
215 std::forward<F>(fn)(*wo.value());
216 return wo.value()->commit();
217 }
218
223 template <typename F>
224 [[nodiscard]] auto with_reader(IdPredicate in_scope, F&& fn)
225 -> expected<void, BackendError>
226 {
227 auto ro = reader(std::move(in_scope));
228 if (!ro) return make_unexpected(std::move(ro.error()));
229 std::forward<F>(fn)(*ro.value());
230 return ro.value()->close();
231 }
232};
