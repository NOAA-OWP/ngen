30 // The mutex serializing calls into the library named @arg Name,
31 // which can't be called from more than one thread at a time. Every
32 // call into that library, anywhere in the process, must hold it:
33 //
34 // std::unique_lock netcdf_lock(multithreading::library_mutex<"netcdf">());
35 //
36 // "netcdf" is the NetCDF library, and the HDF5 library beneath it
37 // unless that was built thread-safe, which can't be called from more
38 // than one thread at a time even for different files.
39 //
40 // It's a plain std::mutex, so regions holding it must not nest.
41 //
42 // Each executable or shared library that calls this gets its own
43 // copy of the reference below, but they all refer to the one mutex
44 // that details::named_library_mutex() keeps for Name. A static
45 // mutex here instead would be one per shared library unless
46 // exported, which GCC won't do for a template specialized on a
47 // string_literal.
48 template <details::string_literal Name>
49 std::mutex& library_mutex()
50 {
51 static std::mutex& mutex = details::named_library_mutex(Name.value);
52 return mutex;
53 }
